# Lane wave4-apollo: Apollo's "execute server command" (mlxrSendExecServerCmd, commonc_ext.h). The
# vendored common-c is upstream's and only ControlStream.c can send on the control stream, so that one
# file is built through a wrapper that includes it and then commonc_servercmd.inc. The submodule is
# not touched; if upstream ever renames what the .inc uses, this configure step says so.
set(MLXR_CC_SRC ${PROJECT_SOURCE_DIR}/third_party/moonlight-common-c/src)
file(READ ${MLXR_CC_SRC}/ControlStream.c MLXR_CC_TEXT)
foreach(needle "static bool sendMessageAndForget(" "static bool encryptedControlStream;"
               "static ENetPeer* peer;" "static ENetHost* client;" "CTRL_CHANNEL_GENERIC")
  string(FIND "${MLXR_CC_TEXT}" "${needle}" MLXR_CC_AT)
  if(MLXR_CC_AT LESS 0)
    message(FATAL_ERROR "moonlight-common-c's ControlStream.c no longer has '${needle}', which stream/commonc_servercmd.inc uses")
  endif()
endforeach()
unset(MLXR_CC_TEXT)
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/ControlStream_mlxr.c @ONLY NEWLINE_STYLE UNIX CONTENT
"// Generated by stream/CMakeLists.txt: moonlight-common-c's ControlStream.c plus the fork's addition
#include \"@MLXR_CC_SRC@/ControlStream.c\"
#include \"@CMAKE_CURRENT_SOURCE_DIR@/commonc_servercmd.inc\"
")
get_target_property(MLXR_CC_SOURCES moonlight-common-c SOURCES)
list(TRANSFORM MLXR_CC_SOURCES REPLACE "^.*src/ControlStream[.]c$" "${CMAKE_CURRENT_BINARY_DIR}/ControlStream_mlxr.c"
     OUTPUT_VARIABLE MLXR_CC_SOURCES_NEW)
list(FIND MLXR_CC_SOURCES_NEW "${CMAKE_CURRENT_BINARY_DIR}/ControlStream_mlxr.c" MLXR_CC_FOUND)
if(MLXR_CC_FOUND LESS 0)
  message(FATAL_ERROR "moonlight-common-c's source list has no src/ControlStream.c to build through the fork's wrapper")
endif()
set_property(TARGET moonlight-common-c PROPERTY SOURCES ${MLXR_CC_SOURCES_NEW})
set_property(SOURCE ${CMAKE_CURRENT_BINARY_DIR}/ControlStream_mlxr.c APPEND PROPERTY OBJECT_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/commonc_servercmd.inc)

add_subdirectory(${PROJECT_SOURCE_DIR}/tests/stream ${CMAKE_CURRENT_BINARY_DIR}/tests)
